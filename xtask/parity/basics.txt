# Basic block behaviour on a frozen superflat world (surface at y=-61).
region -2 -61 -2 12 -52 12

run fill 0 -60 0 10 -60 10 stone
compare

# Shapes follow neighbours: stairs turn into corners, fences connect.
run setblock 2 -59 2 oak_stairs[facing=east]
run setblock 3 -59 2 oak_stairs[facing=north]
run setblock 5 -59 5 oak_fence
run setblock 6 -59 5 oak_fence
run setblock 6 -59 6 cobblestone_wall
compare

# Gravity: sand falls (an entity) a few ticks after it is placed in the air.
run setblock 1 -55 8 sand
step 1
step 20

# Fluids: water spreads by scheduled ticks.
run setblock 8 -59 8 water
step 5
step 40
